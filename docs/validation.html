Release validation

A release fingerprint hashes sorted relative paths, sizes, and file contents. Source and staging accept ordinary local files only. Fresh clones additionally permit git-annex links into their own object store. Frictionless checks both metadata and resource contents; schemas must be inline.

Resource validation uses Frictionless’s built-in multiprocessing by default, including during fresh-retrieval verification. Packages with foreign keys use sequential validation to preserve cross-resource checks. Library callers can also request validate_package(..., parallel=False).

The process pool uses Python’s available CPU count. Set PYTHON_CPU_COUNT before starting Python to limit workers, for example PYTHON_CPU_COUNT=4 uv run depot validate ID VERSION. The Slurm examples set this from SLURM_CPUS_PER_TASK. Allow memory for a decoded table and pandas representation in each worker. Frictionless 5.21 does not forward custom checklists to parallel workers, so parallel validation reports its phase and elapsed time; sequential validation additionally reports individual resources and rows.

Important Prerequisite

Use Spack-managed Python 3.14, Python packages locked by uv, DataLad, git-annex, Git, and Quarto 1.9 or newer. Configure reviewed source roots and publication targets before running release commands.

"""Local-only Frictionless validation and reproducible release fingerprints."""

import hashlib
import json
import os
from pathlib import Path, PurePosixPath
import stat

from frictionless import Check, Checklist, Package

from .config import digest
from .errors import DepotError
from .progress import ProgressCallback, ProgressUpdate, quiet

RESERVED = {".git", ".datalad", ".gitmodules", ".gitattributes"}


class ValidationProgress(Check):
    """Observe Frictionless's normal checks without adding or suppressing errors."""

    type = "depot-progress"

    def __init__(self, on_progress: ProgressCallback, total: int):
        super().__init__()
        self.on_progress = on_progress
        self.total = total
        self.index = 0
        self.rows = 0

    def connect(self, resource):
        super().connect(resource)
        self.index += 1
        self.rows = 0
        self.label = f"resource {self.index}/{self.total}: {resource.name}"
        self.on_progress(ProgressUpdate(f"Loading {self.label}"))

    def validate_start(self):
        self.on_progress(ProgressUpdate(f"Checking {self.label}", 0, unit="rows"))
        yield from ()

    def validate_row(self, row):
        self.rows += 1
        if self.rows % 10_000 == 0:
            self.on_progress(ProgressUpdate(f"Checking {self.label}", self.rows, unit="rows"))
        yield from ()

    def validate_end(self):
        self.on_progress(ProgressUpdate(f"Finished checking {self.label}", self.rows, unit="rows"))
        yield from ()


def release_files(root: Path, annex_root: Path | None = None) -> list[Path]:
    if root.is_symlink() or not root.is_dir():
        raise DepotError("unsafe_release", "Release must be an ordinary directory.")
    paths = []
    for directory, dirs, names in os.walk(root, followlinks=False):
        for name in dirs + names:
            path = Path(directory) / name
            if name in RESERVED or name.endswith((".part", ".partial", ".tmp")):
                raise DepotError(
                    "incomplete_release", "Release contains reserved or unfinished files."
                )
            mode = path.lstat().st_mode
            if stat.S_ISLNK(mode):
                # Only a fresh DataLad clone can use annex links, never source or staging.
                if annex_root is None or not path.resolve().is_relative_to(annex_root.resolve()):
                    raise DepotError("unsafe_path", "Release contains an unauthorized symlink.")
                if not path.is_file():
                    raise DepotError("missing_resource", "Annexed content has not been retrieved.")
            elif not stat.S_ISREG(mode) and not stat.S_ISDIR(mode):
                raise DepotError("unsafe_path", "Release contains a non-regular file.")
        paths.extend(Path(directory) / name for name in names)
    return sorted(paths, key=lambda path: path.relative_to(root).as_posix())


def fingerprint(
    root: Path,
    annex_root: Path | None = None,
    *,
    on_progress: ProgressCallback = quiet,
) -> str:
    on_progress(ProgressUpdate("Inspecting release files"))
    paths = release_files(root, annex_root)
    entries = []
    for index, path in enumerate(paths):
        on_progress(ProgressUpdate("Fingerprinting release files", index, len(paths), "files"))
        before = path.stat()
        checksum = hashlib.sha256()
        with path.open("rb") as stream:
            for block in iter(lambda: stream.read(1024 * 1024), b""):
                checksum.update(block)
        after = path.stat()
        if (before.st_size, before.st_mtime_ns, before.st_ino) != (
            after.st_size,
            after.st_mtime_ns,
            after.st_ino,
        ):
            raise DepotError("source_changed", "A release file changed while being fingerprinted.")
        entries.append([path.relative_to(root).as_posix(), after.st_size, checksum.hexdigest()])
    on_progress(ProgressUpdate("Fingerprinted release files", len(paths), len(paths), "files"))
    return digest(entries)


def local_resource(root: Path, value: object, annex_root: Path | None) -> Path:
    if not isinstance(value, str) or not value or "\\" in value or ":" in value:
        raise DepotError("unsafe_resource", "Only relative local resource paths are supported.")
    relative = PurePosixPath(value)
    if relative.is_absolute() or ".." in relative.parts:
        raise DepotError("unsafe_resource", "Resource path escapes the release.")
    path = root.joinpath(*relative.parts)
    resolved = path.resolve()
    if not resolved.is_relative_to(root.resolve()) and (
        annex_root is None or not resolved.is_relative_to(annex_root.resolve())
    ):
        raise DepotError("unsafe_resource", "Resource path escapes the release.")
    if not path.is_file():
        raise DepotError("missing_resource", "A declared local resource is missing.")
    return path


def validate_package(
    root: Path,
    checks: dict,
    annex_root: Path | None = None,
    *,
    parallel: bool = True,
    on_progress: ProgressCallback = quiet,
) -> dict:
    on_progress(ProgressUpdate("Checking release files and reading datapackage.json"))
    release_files(root, annex_root)
    try:
        descriptor = json.loads((root / "datapackage.json").read_text())
    except (OSError, ValueError) as exc:
        raise DepotError("invalid_package", "Cannot read datapackage.json as JSON.") from exc
    if not isinstance(descriptor, dict):
        raise DepotError("invalid_package", "Package metadata must be an object.")
    for field in checks["required_fields"]:
        if descriptor.get(field) in (None, "", [], {}):
            raise DepotError(
                "missing_metadata", "Required package metadata is missing.", {"field": field}
            )
    resources = descriptor.get("resources")
    if not isinstance(resources, list) or not resources:
        raise DepotError("invalid_package", "At least one local resource is required.")
    summary = []
    for resource in resources:
        if not isinstance(resource, dict):
            raise DepotError("invalid_package", "Each resource must be an object.")
        # Never let descriptor references trigger network reads or local schema escapes.
        for field in ("schema", "dialect", "control"):
            if field in resource and not isinstance(resource[field], dict):
                raise DepotError(
                    "unsafe_resource",
                    "Resource schemas, dialects, and controls must be inline objects.",
                )
        if resource.get("scheme", "file") != "file" or "data" in resource:
            raise DepotError(
                "unsafe_resource", "Resources must declare local files, not remote or inline data."
            )
        paths = resource.get("path")
        paths = paths if isinstance(paths, list) else [paths]
        if not paths:
            raise DepotError("invalid_package", "Each resource must declare at least one path.")
        size = 0
        for value in paths:
            path = local_resource(root, value, annex_root)
            if checks["nonempty"] and path.stat().st_size == 0:
                raise DepotError("empty_resource", "A declared resource is empty.")
            size += path.stat().st_size
        summary.append(
            {"name": resource.get("name", "resource"), "files": len(paths), "bytes": size}
        )
    if "profile" in descriptor and descriptor["profile"] not in {
        "data-package",
        "tabular-data-package",
    }:
        raise DepotError("unsafe_resource", "Only built-in package profiles are supported.")
    try:
        on_progress(ProgressUpdate("Validating package metadata"))
        metadata = Package.validate_descriptor(descriptor, basepath=str(root))
        if not metadata.valid:
            raise DepotError("invalid_package", "Frictionless rejected the package descriptor.")
        package = Package.from_descriptor(descriptor, basepath=str(root))
        # Frictionless requires sequential execution for cross-resource foreign keys.
        run_parallel = parallel and not any(
            resource.schema and resource.schema.foreign_keys for resource in package.resources
        )
        mode = "in parallel" if run_parallel else "sequentially"
        on_progress(ProgressUpdate(f"Validating {len(package.resources)} resources {mode}"))
        checklist = None
        # Frictionless 5.21 does not forward checklists to its parallel workers.
        if not run_parallel and on_progress is not quiet:
            checklist = Checklist(checks=[ValidationProgress(on_progress, len(package.resources))])
        report = package.validate(checklist=checklist, parallel=run_parallel)
        if not report.valid:
            result = report.to_descriptor()
            errors = list(result.get("errors", []))
            for task in result.get("tasks", []):
                errors.extend(task.get("errors", []))
            raise DepotError(
                "invalid_package",
                "Frictionless validation failed.",
                {"error_types": sorted({error["type"] for error in errors})},
            )
        minimum = checks.get("minimum_rows", 0)
        for task in report.to_descriptor().get("tasks", []):
            if minimum and task.get("stats", {}).get("rows", 0) < minimum:
                raise DepotError("insufficient_rows", "Resource has fewer rows than required.")
    except DepotError:
        raise
    except Exception as exc:
        raise DepotError(
            "invalid_package", "Frictionless could not validate this package."
        ) from exc
    on_progress(ProgressUpdate("Package validation passed"))
    return {"valid": True, "resources": summary, "package_name": descriptor.get("name", "")}
