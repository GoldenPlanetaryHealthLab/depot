DataLad publication

Each dataset has one managed DataLad repository containing releases/VERSION and a local RIA archive at TARGET_ROOT/ID. DataLad configures a Git sibling plus an annex storage sibling. A RIA archive is a bare repository, so unlocking files in the publisher’s workspace cannot make ordinary files appear in Goldmine. See DataLad’s RIA layout.

The default access = "unlocked" also exposes an independent ordinary-file view at TARGET_ROOT/ID.data/releases/VERSION. After retrieval and validation, Depot copies the verified bytes there atomically and checks their fingerprint. The files are populated, writable by their owner, and contain no annex symlinks or placeholders. Group access follows filesystem permissions and the installation’s umask. There is one extra full copy of the data; no hardlinks can let user edits alter archived objects. Personal analysis copies keep the shared view intact.

Set target access = "locked" in TOML, or override with destination.access in a dataset’s registry YAML, to retain archive-only publication. New annex files in the managed workspace follow that access setting through the supported annex.addunlocked option. A fresh consumer clone still requires datalad get, regardless of pointer mode.

Repeat pushes reconcile missing views from staged bytes matching the previously verified fingerprint, retaining the original archived commit without another Frictionless pass. Changed ordinary files cause an access_conflict; they are never overwritten. A failed first materialization leaves no successful receipt and can be retried. Previous successful archive receipts remain intact. Switching to locked mode skips materialization; it does not delete an existing ordinary copy or retroactively lock earlier commits. No parent dataset is assumed.

Important Prerequisite

Use Spack-managed Python 3.14, Python packages locked by uv, DataLad, git-annex, Git, and Quarto 1.9 or newer. Configure reviewed source roots and publication targets before running release commands.

publication.py

"""DataLad CLI publication to local RIA stores, with fresh-clone verification."""

from collections.abc import Callable
from pathlib import Path
import shutil
import subprocess
from tempfile import TemporaryDirectory

from .config import Config, access_mode, contained
from .errors import DepotError
from .progress import ProgressCallback, ProgressUpdate, quiet
from .validation import fingerprint, validate_package


def command(args: list[str], cwd: Path | None = None) -> str:
    try:
        result = subprocess.run(args, cwd=cwd, check=True, capture_output=True, text=True)
        return result.stdout.strip()
    except FileNotFoundError as exc:
        raise DepotError(
            "missing_tool", f"Install {args[0]} and make it available on PATH."
        ) from exc
    except subprocess.CalledProcessError as exc:
        # Raw stdout/stderr can contain credentials, remote URLs, and private paths.
        raise DepotError(
            "command_failed",
            f"{args[0]} command failed; check tools, target access, and permissions.",
            {"operation": args[1], "exit_code": exc.returncode},
        ) from exc


def materialize(
    config: Config,
    manifest: dict,
    version: str,
    verified: Path,
    expected: str,
    *,
    on_progress: ProgressCallback = quiet,
) -> dict:
    """Expose verified bytes as independent ordinary files; never overwrite edits."""
    mode = access_mode(config, manifest)
    if mode == "locked":
        return {"mode": mode, "populated": False}
    target = config.targets[manifest["destination"]["target"]]
    relative = Path(manifest["id"] + ".data") / "releases" / version
    destination = target.root / relative
    # Reject redirected views even when a symlink stays within the target root.
    for path in (destination, *destination.parents):
        if path == target.root:
            break
        if path.is_symlink():
            raise DepotError(
                "unsafe_path", "The ordinary-file view cannot contain directory symlinks."
            )
    destination = contained(destination, target.root)
    on_progress(ProgressUpdate("Preparing populated, unlocked Goldmine files"))
    if destination.exists():
        if fingerprint(destination, on_progress=on_progress) != expected:
            raise DepotError(
                "access_conflict",
                "Goldmine ordinary files have changed; preserve edits and restore or relocate the view before retrying.",
            )
    else:
        destination.parent.mkdir(parents=True, exist_ok=True)
        with TemporaryDirectory(prefix=".populate-", dir=destination.parent) as temporary:
            incoming = Path(temporary) / "release"
            # Follow only the annex links already checked by retrieval verification.
            # copyfile creates independent bytes, without inheriting annex read-only modes.
            shutil.copytree(verified, incoming, copy_function=shutil.copyfile)
            if fingerprint(incoming, on_progress=on_progress) != expected:
                raise DepotError(
                    "verification_failed", "Populated files differ from verified content."
                )
            incoming.replace(destination)
    for path in destination.rglob("*"):
        if path.is_file():
            path.chmod(path.stat().st_mode | 0o200)
    return {
        "mode": mode,
        "populated": True,
        "path": str(destination),
        "relative_path": relative.as_posix(),
    }


def publish(
    config: Config,
    manifest: dict,
    version: str,
    staged: Path,
    expected: str,
    on_saved: Callable[[str], None] | None = None,
    on_progress: ProgressCallback = quiet,
) -> dict:
    on_progress(ProgressUpdate("Preparing DataLad publication workspace"))
    dataset_id = manifest["id"]
    dataset = config.dataset_path(config.workspace, dataset_id)
    target = config.targets[manifest["destination"]["target"]]
    store = contained(target.root / dataset_id, target.root)
    store_url = "ria+" + store.as_uri()
    clone_url = store_url + "#~" + dataset_id
    config.workspace.mkdir(parents=True, exist_ok=True)
    if not (dataset / ".datalad/config").is_file():
        command(["datalad", "create", "-d", str(dataset), "-D", f"Depot dataset {dataset_id}"])
    command(
        [
            "git",
            "config",
            "annex.addunlocked",
            str(access_mode(config, manifest) == "unlocked").lower(),
        ],
        cwd=dataset,
    )
    destination = contained(dataset / "releases" / version, dataset)
    if destination.exists():
        # On retry the saved worktree may contain annex symlinks.
        if (
            fingerprint(destination, dataset / ".git/annex/objects", on_progress=on_progress)
            != expected
        ):
            raise DepotError(
                "version_conflict", "Managed workspace content conflicts with this version."
            )
    else:
        destination.parent.mkdir(exist_ok=True)
        # A failed copy must not strand a partial version under its final name.
        with TemporaryDirectory(prefix=".intake-", dir=dataset) as temporary:
            incoming = Path(temporary) / "release"
            on_progress(ProgressUpdate("Copying validated release into DataLad workspace"))
            shutil.copytree(staged, incoming)
            if fingerprint(incoming, on_progress=on_progress) != expected:
                raise DepotError("source_changed", "Staged content changed during publication.")
            incoming.replace(destination)
    on_progress(ProgressUpdate("Saving release with DataLad"))
    command(
        [
            "datalad",
            "save",
            "-d",
            str(dataset),
            "-m",
            f"Depot release {version}",
            str(destination),
        ],
        cwd=dataset,
    )
    # Store the release's own commit, rather than a later dataset HEAD on retry.
    commit = command(["git", "log", "-1", "--format=%H", "--", f"releases/{version}"], cwd=dataset)
    if not commit:
        raise DepotError("invalid_state", "No DataLad commit exists for the release.")
    if on_saved:
        on_saved(commit)
    configured = command(["git", "remote"], cwd=dataset).splitlines()
    if "goldmine" not in configured:
        on_progress(ProgressUpdate("Creating publication storage sibling"))
        target.root.mkdir(parents=True, exist_ok=True)
        command(
            [
                "datalad",
                "create-sibling-ria",
                "-d",
                str(dataset),
                "-s",
                "goldmine",
                "--alias",
                dataset_id,
                "--new-store-ok",
                store_url,
            ],
            cwd=dataset,
        )
    else:
        remote = command(["git", "remote", "get-url", "goldmine"], cwd=dataset)
        if not Path(remote).resolve().is_relative_to(store):
            raise DepotError(
                "target_conflict", "Existing sibling belongs to a different publication target."
            )
    on_progress(ProgressUpdate("Pushing release to publication storage"))
    command(
        ["datalad", "push", "-d", str(dataset), "--to", "goldmine", "--data", "anything"],
        cwd=dataset,
    )
    with TemporaryDirectory(prefix="verify-", dir=config.staging) as temporary:
        on_progress(ProgressUpdate("Cloning published dataset for retrieval verification"))
        clone = Path(temporary) / "clone"
        command(["datalad", "clone", clone_url, str(clone)])
        command(["git", "checkout", "--detach", commit], cwd=clone)
        # Explicitly choose the RIA storage sibling: publisher-local content cannot
        # accidentally satisfy the verification boundary.
        on_progress(ProgressUpdate("Retrieving published release from storage"))
        command(
            [
                "datalad",
                "get",
                "-d",
                str(clone),
                "--source",
                "goldmine-storage",
                str(clone / "releases" / version),
            ],
            cwd=clone,
        )
        retrieved = clone / "releases" / version
        on_progress(ProgressUpdate("Verifying retrieved release fingerprint"))
        if (
            fingerprint(retrieved, clone / ".git/annex/objects", on_progress=on_progress)
            != expected
        ):
            raise DepotError(
                "verification_failed", "Fresh retrieval does not match the validated content."
            )
        on_progress(ProgressUpdate("Validating retrieved release"))
        validate_package(
            retrieved, manifest["checks"], clone / ".git/annex/objects", on_progress=on_progress
        )
        access = materialize(
            config, manifest, version, retrieved, expected, on_progress=on_progress
        )
    on_progress(ProgressUpdate("Publication and fresh retrieval verified"))
    return {
        "commit": commit,
        "target": str(store),
        "clone_url": clone_url,
        "verified": True,
        "access": access,
    }
